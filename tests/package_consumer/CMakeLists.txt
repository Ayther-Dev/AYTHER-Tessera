cmake_minimum_required(VERSION 3.25)

if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
project(TesseraInstalledConsumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT TESSERA_TEST_PREFIX)
    message(FATAL_ERROR "PACKAGE_TEST_PREFIX_REQUIRED: falta el prefijo instalado.")
endif()
if(NOT TESSERA_TEST_FIND_VERSION)
    set(TESSERA_TEST_FIND_VERSION 0.1)
endif()
if(TESSERA_TEST_BAD_ARCH)
    set(CMAKE_CXX_COMPILER_ARCHITECTURE_ID ARM64)
endif()
if(TESSERA_TEST_BAD_TOOLSET)
    set(MSVC_VERSION 1944)
endif()
find_package(Tessera ${TESSERA_TEST_FIND_VERSION} CONFIG REQUIRED
    PATHS "${TESSERA_TEST_PREFIX}" NO_DEFAULT_PATH)
if(NOT TARGET Tessera::core)
    message(FATAL_ERROR "PACKAGE_TARGET_MISSING: falta Tessera::core.")
endif()
if(NOT Tessera_ABI_MAJOR EQUAL 1 OR NOT Tessera_ABI_MINOR EQUAL 0 OR
        NOT Tessera_POINTER_BYTES EQUAL 8 OR
        NOT Tessera_MSVC_RUNTIME STREQUAL "MultiThreadedDLL" OR
        NOT Tessera_VERSION VERSION_EQUAL "0.1.0")
    message(FATAL_ERROR "PACKAGE_METADATA_INVALID: versión, ABI o runtime incompatibles.")
endif()

# An installed static target must not pull private source-only helper targets.
foreach(property INTERFACE_LINK_LIBRARIES INTERFACE_INCLUDE_DIRECTORIES
        INTERFACE_LINK_OPTIONS INTERFACE_COMPILE_OPTIONS)
    get_target_property(value Tessera::core ${property})
    if(value)
        if(value MATCHES "PicoSHA2|build_options|tessera_picosha2|nlohmann|doctest")
            message(FATAL_ERROR "PACKAGE_PRIVATE_TARGET_LEAK: ${property}=${value}")
        endif()
        string(REPLACE "${TESSERA_TEST_PREFIX}" "<installed-prefix>" inspected "${value}")
        foreach(forbidden IN ITEMS "${TESSERA_TEST_SOURCE_PATH}"
                "${TESSERA_TEST_BUILD_PATH}" "${TESSERA_TEST_ORIGINAL_PREFIX}")
            if(forbidden STREQUAL "")
                continue()
            endif()
            string(FIND "${inspected}" "${forbidden}" position)
            if(NOT position EQUAL -1)
                message(FATAL_ERROR "PACKAGE_SOURCE_PATH_LEAK: ${property}=${value}")
            endif()
        endforeach()
    endif()
endforeach()

add_executable(tessera_installed_consumer main.cpp abi_c.c)
target_link_libraries(tessera_installed_consumer PRIVATE Tessera::core)
if(Tessera_DIAGNOSTICS)
    if(NOT COMMAND tessera_copy_runtime_dependencies)
        message(FATAL_ERROR "PACKAGE_ASAN_HELPER_MISSING: falta el helper de runtime instalado.")
    endif()
    tessera_copy_runtime_dependencies(tessera_installed_consumer)
endif()
enable_testing()
add_test(NAME installed.c_and_cpp_abi COMMAND tessera_installed_consumer)
